% Per-author note macros: three lines each.  The notecolor-<name> colorlet is
% load-bearing --- \claude[<name>] and friends derive the owner styling from it.
\colorlet{notecolor-jac}{green!20}
\newcommand{\jac}[2][]{\note[#1]{jac}{notecolor-jac}{#2}}
\newcommand{\Jac}[2][]{\jac[inline,#1]{#2}}

\colorlet{notecolor-timv}{magenta!20}
\newcommand{\timv}[2][]{\note[#1]{timv}{notecolor-timv}{#2}}
\newcommand{\Timv}[2][]{\timv[inline,#1]{#2}}

\colorlet{notecolor-ryan}{violet!20}
\newcommand{\ryan}[2][]{\note[#1]{ryan}{notecolor-ryan}{#2}}
\newcommand{\Ryan}[2][]{\ryan[inline,#1]{#2}}
