% example.tex --- demonstrates every note style in mytodonotes + agentnotes
%
% Build everything in this directory with `make`, or one at a time (latexmkrc
% here puts the repo's own .sty files first on TEXINPUTS):
%   latexmk -pdf example.tex
% Parskip variant (paragraphs separated by skips instead of indents):
%   latexmk -pdf -jobname=example-parskip -usepretex='\def\exampleparskip{1}' example.tex
\documentclass{article}
\usepackage[left=1in,right=2.9in,top=1in,bottom=1in]{geometry}
\ifdefined\exampleparskip
  \usepackage{parskip}
\fi
\usepackage{mytodonotes}
\usepackage{agentnotes}
\setlength{\marginparwidth}{5.4cm}

% One \newagent line per agent working on the paper.  A stem with a built-in
% preset needs nothing else; anything unlisted supplies its own name, color and
% mark.  Several agents coexist without collision, so a paper where each author
% brings their own assistant works as well as one where everybody shares Claude.
\newagent{claude}
\newagent{codex}
\newagent{gemini}
\newagent{ai}
\newagent{qwen}[name=Qwen, color=orange!60, mark=\ding{92}]

% Per-author note macros: three lines each.  Every agent derives its owner
% styling from the notecolor-<name> colorlet, so defining one is what lets
% \claude[<name>], \codex[<name>] and the rest pick up that author's color.
\colorlet{notecolor-alice}{green!20}
\newcommand{\alice}[2][]{\note[#1]{alice}{notecolor-alice}{#2}}
\newcommand{\Alice}[2][]{\alice[inline,#1]{#2}}

\colorlet{notecolor-bob}{magenta!20}
\newcommand{\bob}[2][]{\note[#1]{bob}{notecolor-bob}{#2}}
\newcommand{\Bob}[2][]{\bob[inline,#1]{#2}}

\colorlet{notecolor-carol}{violet!20}
\newcommand{\carol}[2][]{\note[#1]{carol}{notecolor-carol}{#2}}
\newcommand{\Carol}[2][]{\carol[inline,#1]{#2}}
